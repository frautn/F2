\begin{Exercise}\label{p:continuas01}
  \textbf{\raisebox{.5pt}{\textcircled{\raisebox{-1.2pt} {E}}}} \textit{a}) Para el segmento mostrado en la figura \ref{f:continuas01}, de longitud $L$ y carga $Q$ distribuida uniformemente, demuestre que el campo eléctrico en el punto $S$ ubicado a una distancia $a$ sobre la mediatriz, está dado por:
  \begin{align*}
    \va*{E} &= \frac{2kQ}{a} \frac{1}{\sqrt{L^2+4a^2}}\,\vu{y}\,.
  \end{align*}
  \textit{b}) Compruebe que si el segmento es muy largo (\textsl{hilo infinito}), siendo $L \gg a$, el campo en el punto $S$ se puede aproximar por
  \begin{align*}
    \va*{E} &= \frac{2k\lambda}{a}\,\vu{y}\,,
  \end{align*}
  donde $\lambda = \frac{Q}{L}$ es la densidad lineal de carga.\par\medskip
  \textit{c}) Para el mismo segmento, demuestre que el campo eléctrico en el punto $P$ vale:
  \begin{align*}
    \va*{E} = \frac{kQ}{L} \left \lbrack \frac{1}{b} - \frac{1}{L+b} \right \rbrack \,\vu{x}\, .
  \end{align*}
\end{Exercise}

\begin{Exercise}\label{p:continuas03}
Se tienen dos hilos infinitos con densidad lineal de carga uniforme, ubicados como muestra la figura \ref{f:continuas03}. Uno de los hilos se prolonga a lo largo del eje $y$ ($x=0$) y tiene densidad de carga $\lambda_1 = \SI{-30}{\micro\coulomb/\metre}$, y el otro hilo se encuentra a una distancia $b = \SI{20}{\centi\metre}$ medida sobre el eje $x$, y su densidad de carga es $\lambda_2 = \SI{10}{\micro\coulomb/\metre}$.\par\medskip
\textit{a}) Calcular el vector campo eléctrico resultante en la posición $(15;0;0)\text{ cm}$.\par\medskip 
\textit{b}) Calcular el vector campo eléctrico resultante en la posición $(0;0;10)\text{ cm}$. \par\medskip
\textit{c}) ¿En qué posiciones sobre el eje $x$ se puede ubicar una carga puntual positiva si se desea que la fuerza neta sobre dicha carga esté dirigida hacia el sentido positivo de $x$?
\end{Exercise}
\begin{Answer}
  \begin{minipage}[t]{.4\textwidth}
    \textit{a}) $\va*{E}(0.15;0;0) = \SI{-7.2E6}{N/C}\,\vu{x}$\\
    \textit{b}) $\va*{E}(0;0;0.10) = \SI{-0.72E6}{N/C}\,\vu{x} - \SI{3.24E6}{N/C}\,\vu{z}$\\
    \textit{c}) La carga puede estar en $ x < \SI{0}{\centi\metre}$ \\ o en $\SI{20}{\centi\metre} < x < \SI{30}{\centi\metre}.$
  \end{minipage}
\end{Answer}
